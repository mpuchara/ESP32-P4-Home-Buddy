idf_component_register(SRCS "face_policy.c" "face_worker.cpp" INCLUDE_DIRS "include"
 REQUIRES esp_common PRIV_REQUIRES camera compute_budget esp_timer esp_psram espressif__human_face_detect)
idf_component_get_property(model_dir espressif__human_face_detect COMPONENT_DIR)
foreach(item IN ITEMS msr mnp)
 file(SHA256 "${model_dir}/models/p4/human_face_detect_${item}_s8_v1.espdl" digest)
 if(item STREQUAL "msr")
  set(expected "2bbaaf17f39ce9ee547abf5ffdc6d8fcd1844009a0d26878d7bfb33451f16211")
 else()
  set(expected "5b7fedcff59dbe69d1935d6d206bf371fc18b5dd9176320ef69c9f05ee83af55")
 endif()
 if(NOT digest STREQUAL expected)
  message(FATAL_ERROR "Unexpected human face detection weights: ${item}")
 endif()
endforeach()
